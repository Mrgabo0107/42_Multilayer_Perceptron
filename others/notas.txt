        3. ¿La Loss depende de la activación?Aquí hay un secreto de "matemática elegante" 
        que te servirá para los bonus: La combinación Softmax + Categorical Cross-Entropy.
        Si intentas calcular la derivada de Softmax por un lado y la de Categorical Cross-Entropy 
        por otro, la matemática es un poco pesadilla. Pero, cuando las combinas para sacar el 
        gradiente de la última capa, la fórmula se simplifica milagrosamente


         a:$$\frac{\partial Loss}{\partial z} = \hat{y} - y$$
         (Donde $\hat{y}$ es tu predicción y $y$ es el valor real).

         Es decir, el gradiente es simplemente la resta entre lo que dijiste y la verdad.
          Por eso se usan juntas; son el "matrimonio perfecto" matemático.

import json
import sys

def validate_config(config_path, input_dim):
    try:
        with open(config_path, 'r') as f:
            conf = json.load(f)
    except Exception as e:
        print(f"Error reading JSON: {e}")
        sys.exit(1)

    # 1. Validar parámetros de entrenamiento
    train_params = conf.get("training", {})
    if train_params.get("learning_rate", 0) <= 0:
        raise ValueError("learning_rate must be positive.")
    if train_params.get("batch_size", 0) <= 0:
        raise ValueError("batch_size must be at least 1.")

    # 2. Validar Topología
    topology = conf.get("topology", [])
    if len(topology) < 3: # Entrada + 2 ocultas (mínimo según subject)
        raise ValueError("Network must have at least two hidden layers.")

    for i, layer in enumerate(topology):
        # Validar n_neurons
        n = layer.get("n_neurons")
        if not isinstance(n, int) or n <= 0:
            raise ValueError(f"Layer {i} must have a positive integer of neurons.")
        
        # Validar activaciones
        act = layer.get("activation")
        valid_acts = ["relu", "sigmoid", "tanh", "softmax"]
        if act not in valid_acts:
            raise ValueError(f"Activation '{act}' not supported. Use: {valid_acts}")
        
        # Regla de oro: Softmax solo al final
        if act == "softmax" and i != len(topology) - 1:
            raise ValueError("Softmax activation is only allowed in the output layer.")
        
        # Validar inicializadores
        init = layer.get("initializer")
        valid_inits = ["heUniform", "xavier", "random"]
        if init not in valid_inits:
            raise ValueError(f"Initializer '{init}' not supported. Use: {valid_inits}")

    # 3. Validar consistencia con el dataset
    if topology[-1]["n_neurons"] != 2:
        raise ValueError("The output layer must have exactly 2 neurons for M/B classification.")

    return conf
